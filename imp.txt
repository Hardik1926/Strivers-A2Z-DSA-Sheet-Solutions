Scanner scan = new Scanner(System.in);
        
        System.out.print("Enter the no of elements in Array: ");
        int n = scan.nextInt();

        int[] nums = new int[n];
        
        for(int i = 0; i<n;i++){
            System.out.print("Enter the Elements: ");
            int x = scan.nextInt();
            nums[i] = x ;
        }
        scan.close();

        System.out.print("[");
        for(int i = 0; i < nums.length;i++){
            System.out.print(nums[i]);
            if(i < nums.length - 1 ){
                System.out.print(",");
            }    
        }    
        System.out.print("]");
